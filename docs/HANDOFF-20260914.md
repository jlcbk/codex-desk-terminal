```bash
# Codex 模式（当前）：
pkill -f bridge_serve; sleep 2
nohup uv run --python 3.12 --with websockets==17.1 python3 scripts/bridge_serve_codex.py serve \
  --host 192.168.1.124 --port 8765 --cert config/local/dev-certs/server.crt \
  --key config/local/dev-certs/server.key --token-file config/local/device_token \
  > artifacts/bridge/codex_serve_live.log 2>&1 & disown
# 提交任务：uv run --python 3.12 --with websockets==17.1 \
#   python3 scripts/bridge_serve_codex.py submit --prompt "任务"

# —— 回归（全量树口径务必带 psutil）——

  python -m pytest tests/ -q          # 当前基线 398 passed
sh scripts/build_shared.sh && sh scripts/build_presenter_tests.sh && sh scripts/build_transport_tests.sh
uv run --python 3.12 --with pillow python3 scripts/check_ui.py \
  --golden tests/golden --actual artifacts/ui/replay   # 22/22 零差异
# golden 重生成（仅 UI 有意变更后，A0 目检）：scripts/gen_goldens.py --golden tests/golden --force
# ⚠重生成前必须先 cmake --build build/simulator（旧二进制会产假绿）

# —— 烧录（板 1301）——
. "$HOME/esp/esp-idf-v5.5.5/export.sh" >/dev/null 2>&1 && (cd firmware && idf.py build && idf.py -p /dev/cu.usbmodem1301 flash)
```
